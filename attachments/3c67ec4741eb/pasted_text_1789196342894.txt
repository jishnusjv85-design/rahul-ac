        function setBookingStep(step) {
            currentBookingStep = step
            
            document.getElementById("booking-step-1").classList.add("hidden")
            document.getElementById("booking-step-2").classList.add("hidden")
            document.getElementById("booking-step-3").classList.add("hidden")
            
            document.getElementById(`booking-step-${step}`).classList.remove("hidden")
            
            // update dots
            for (let i = 1; i <= 3; i++) {
                const dot = document.getElementById(`step-${i}-dot`)
                if (i < step) {
                    dot.classList.add("bg-emerald-400", "text-white")
                    dot.classList.remove("bg-white/20")
                } else if (i === step) {
                    dot.classList.add("bg-orange-500")
                    dot.classList.remove("bg-white/20")
                } else {
                    dot.classList.remove("bg-orange-500", "bg-emerald-400")
                    dot.classList.add("bg-white/20")
                }
            }
            
            if (step === 2) {
                renderDateScroller()
                renderTimeSlots()
            }
            
            // update summary live
            updateLiveSummary()
        }

        function renderBookingsList() {
            const container = document.getElementById("bookings-list")

                if (!currentUser) return true
                return b.email === currentUser.email
            })
            
            if (userBookings.length === 0) {
                document.getElementById("no-bookings").classList.remove("hidden")
                return
            } else {
                document.getElementById("no-bookings").classList.add("hidden")
            }
            
            userBookings.forEach(booking => {
                const statusColor = booking.status === "confirmed" ? "emerald" : (booking.status === "completed" ? "violet" : "amber")
                
                const div = document.createElement("div")
                div.className = "glass border border-white/10 rounded-3xl p-6 flex gap-6 items-center"
                div.innerHTML = `
                    <div class="w-12 h-12 text-5xl flex items-center justify-center">${booking.icon}</div>
                    <div class="flex-1">
                        <div class="flex justify-between">
                            <div>
                                <span class="font-semibold text-xl">${booking.service}</span>
                                <span class="ml-4 text-xs px-4 py-1 bg-${statusColor}-400 text-zinc-900 rounded-3xl">${booking.status.toUpperCase()}</span>
                            </div>
                            <div class="font-mono text-xs text-white/30">${booking.id}</div>
                        </div>
                        <div class="text-xs text-white/40 mt-px">${booking.date} • ${booking.time}</div>
                        <div class="text-xs text-zinc-400 mt-4">${booking.description}</div>
                    </div>
                    <div>
                        <button onclick="cancelBooking('${booking.id}'); event.stopImmediatePropagation()" 
                                class="text-xs px-5 py-4 border border-white/30 rounded-3xl text-white/60 hover:text-rose-400">CANCEL</button>
                    </div>

        function renderAdminTable(filter = "all") {
            const tbody = document.getElementById("admin-table-body")
            tbody.innerHTML = ""
            
            let filtered = bookings
            
            if (filter === "pending") {
                filtered = bookings.filter(b => b.status === "pending")
            } else if (filter === "confirmed") {
                filtered = bookings.filter(b => b.status === "confirmed")
            }
            
            filtered.forEach((booking, index) => {
                const row = document.createElement("tr")
                row.className = "table-row border-b border-white/10 last:border-none hover:bg-white/5"
                row.innerHTML = `
                    <td class="px-8 py-7 font-mono text-xs text-white/40">${booking.id}</td>
                    <td class="px-8 py-7">
                        <div class="font-medium">${booking.user}</div>
                        <div class="text-xs text-white/30">${booking.email}</div>
                    </td>
                    <td class="px-8 py-7">
                        <div class="flex items-center gap-x-3">
                            <span class="text-3xl">${booking.icon}</span>
                            <span>${booking.service}</span>
                        </div>
                    </td>
                    <td class="px-8 py-7 text-sm">${booking.date}<br><span class="font-light text-xs">${booking.time}</span></td>
                    <td class="px-8 py-7 font-medium text-sky-300">${booking.phone}</td>
                    <td class="px-8 py-7">
                        <span onclick="updateBookingStatus(${index}); event.stopImmediatePropagation()" 
                              class="status-${booking.status} text-xs font-bold px-6 py-2 rounded-3xl">${booking.status}</span>
                    </td>
                    <td class="px-8 py-7 text-right">
                        <button onclick="viewBookingDetail('${booking.id}');event.stopImmediatePropagation()" 
                                class="text-xs bg-white/10 px-7 py-3.5 rounded-3xl">DETAILS</button>
                    </td>
                `
                tbody.append(row)
            })
            
            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="px-8 py-12 text-center text-white/30">No bookings match the filter</td></tr>`
            }
        }
