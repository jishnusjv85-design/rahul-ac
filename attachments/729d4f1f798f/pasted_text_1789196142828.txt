    function localDateISO(date) {
      const y = date.getFullYear();
      const m = String(date.getMonth() + 1).padStart(2, "0");
      const d = String(date.getDate()).padStart(2, "0");
      return `${y}-${m}-${d}`;
    }

    function futureDate(days) {
      const date = new Date();
      date.setDate(date.getDate() + days);
      return localDateISO(date);
    }

    function seedRequests() {

    function escapeHTML(value) {
      return String(value || "").replace(/[&<>"']/g, char => ({
        "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#039;"
      })[char]);
    }

    function renderNav() {
      const actions = document.getElementById("navActions");

      if (!session) {

    function scrollToBooking() {
      document.getElementById("booking").scrollIntoView({ behavior: "smooth", block: "start" });
    }

    function prefillService(service) {

      scrollToBooking();
    }

    function selectService(service) {
      booking.service = service;
      document.querySelectorAll("[data-service]").forEach(el => {
        el.classList.toggle("selected", el.dataset.service === service);
      });
      updateSummary();
    }

    function selectDuration(duration) {
      booking.duration = duration;
      document.querySelectorAll("[data-duration]").forEach(el => {
        el.classList.toggle("selected", el.dataset.duration === duration);
      });
      updateSummary();
    }

    function setDate(value) {
      booking.date = value;
      if (booking.time && isSlotPast(booking.time)) booking.time = "";
      renderTimeSlots();
      updateSummary();
    }

    function parseTime(time) {
      const [clock, period] = time.split(" ");
      let [hour, minute] = clock.split(":").map(Number);
      if (period === "PM" && hour !== 12) hour += 12;
      if (period === "AM" && hour === 12) hour = 0;
      return { hour, minute };
    }

    function isSlotPast(slot) {
      if (!booking.date || booking.date !== localDateISO(new Date())) return false;
      const now = new Date();
      const parsed = parseTime(slot);
      const appointment = new Date();
      appointment.setHours(parsed.hour, parsed.minute, 0, 0);
      return appointment <= now;
    }

    function renderTimeSlots() {
      const grid = document.getElementById("timeGrid");
      grid.innerHTML = slots.map(slot => {
        const disabled = isSlotPast(slot);
        return `<button class="option-button ${booking.time === slot ? "selected" : ""}" type="button" ${disabled ? "disabled" : ""} onclick="selectTime('${slot}')">${slot}</button>`;
      }).join("");
    }

    function selectTime(time) {
      booking.time = time;
      renderTimeSlots();
      updateSummary();
    }

    function formatDate(value) {
      if (!value) return "Not selected";
      return new Intl.DateTimeFormat("en-US", {
        weekday:"short", month:"short", day:"numeric", year:"numeric"
      }).format(new Date(value + "T12:00:00"));
    }

    function updateSummary() {
      document.getElementById("summaryService").textContent = booking.service || "Not selected";
      document.getElementById("summaryDuration").textContent = booking.duration || "Not selected";
      document.getElementById("summaryDate").textContent = formatDate(booking.date);
      document.getElementById("summaryTime").textContent = booking.time || "Not selected";
    }

    function setStep(step) {
      currentStep = step;
      document.querySelectorAll(".form-step").forEach(el => {
        el.classList.toggle("active", Number(el.dataset.step) === step);
      });
      document.querySelectorAll(".progress-item").forEach((el, index) => {
        el.classList.toggle("active", index + 1 <= step);
      });
    }

    function nextStep() {
      if (currentStep === 1 && !booking.service) return showToast("Please select a service category.");
      if (currentStep === 2 && !booking.duration) return showToast("Please select the expected work time.");
      if (currentStep === 3 && (!booking.date || !booking.time)) return showToast("Please choose a convenient date and time.");
      if (currentStep < 4) setStep(currentStep + 1);
    }

    function previousStep() {
      if (currentStep > 1) setStep(currentStep - 1);
    }

    function validPhone(phone) {
      return phone.replace(/\D/g, "").length >= 10;
    }

    function submitBooking(event) {
      event.preventDefault();
      booking.phone = document.getElementById("phone").value.trim();
      booking.area = document.getElementById("area").value.trim();
      booking.notes = document.getElementById("notes").value.trim();

      if (!booking.service || !booking.duration || !booking.date || !booking.time) {
        showToast("Please complete all booking steps.");
        return;
      }

      if (!validPhone(booking.phone)) {
        showToast("Please enter a valid phone number.");
        document.getElementById("phone").focus();
        return;
      }

      if (!booking.area) {
        showToast("Please enter the service location.");
        return;
      }

      if (!session || session.role !== "user") {
        pendingSubmission = true;
        openGmailLogin();
        return;
      }

      createRequest();
    }

    function createRequest() {
      const request = {
        id: "FX-" + Math.floor(10000 + Math.random() * 89999),
        name: session.name,
        email: session.email,
        service: booking.service,
        duration: booking.duration,
        date: booking.date,
        time: booking.time,
        phone: booking.phone,
        area: booking.area,
        notes: booking.notes,
        status: "Request received",
        created: Date.now()
      };

      const requests = getRequests();
      requests.unshift(request);
      saveRequests(requests);

      document.getElementById("successDetails").innerHTML = `
        <div><span>Request ID</span><strong>${escapeHTML(request.id)}</strong></div>
        <div><span>Service</span><strong>${escapeHTML(request.service)}</strong></div>
        <div><span>Appointment</span><strong>${escapeHTML(formatDate(request.date))} · ${escapeHTML(request.time)}</strong></div>
        <div><span>Contact</span><strong>${escapeHTML(request.phone)}</strong></div>
      `;

      openModal("successModal");
      pendingSubmission = false;
    }

    function resetBooking() {
      Object.assign(booking, { service:"", duration:"", date:"", time:"", notes:"", phone:"", area:"" });
      document.getElementById("bookingForm").reset();
      document.querySelectorAll(".selected").forEach(el => el.classList.remove("selected"));
      setStep(1);
      renderTimeSlots();
      updateSummary();
      scrollToBooking();
    }

    function openGmailLogin() {
      if (session?.role === "admin") {
        showToast("Admin accounts have read-only request access.");

      openModal("gmailModal");
      setTimeout(() => document.getElementById("gmailName").focus(), 100);

    function signInWithGmail(event) {
      event.preventDefault();
      const name = document.getElementById("gmailName").value.trim();
      const email = document.getElementById("gmailEmail").value.trim().toLowerCase();
      const error = document.getElementById("gmailError");
      const valid = /^[^\s@]+@gmail\.com$/i.test(email);

      error.classList.toggle("show", !valid);
      if (!valid || !name) return;

      session = { role:"user", name, email };
      localStorage.setItem("fixly_session", JSON.stringify(session));
      closeModal("gmailModal");
      renderNav();
      showToast(`Welcome, ${name.split(" ")[0]}!`);

      if (pendingSubmission) createRequest();
    }

    function adminSignIn(event) {
      event.preventDefault();
      const email = document.getElementById("adminEmail").value.trim().toLowerCase();
      const code = document.getElementById("adminCode").value;
      const valid = email === "admin@fixly.team" && code === "246810";
      document.getElementById("adminError").classList.toggle("show", !valid);
      if (!valid) return;

      session = { role:"admin", name:"Fixly Admin", email };
      localStorage.setItem("fixly_session", JSON.stringify(session));
      closeModal("adminLoginModal");
      renderNav();
      openAdminDashboard();
    }

      const normalized = query.trim().toLowerCase();

        return !normalized || Object.values(request).some(value =>

      if (!requests.length) {
        target.innerHTML = `<div class="empty-state">No matching requests found.</div>`;
        return;

      target.innerHTML = `

    function openModal(id) {
      closeAllModals();
      const modal = document.getElementById(id);
      modal.classList.add("open");
      modal.setAttribute("aria-hidden", "false");
      document.body.classList.add("modal-open");
    }

    function closeModal(id) {
      const modal = document.getElementById(id);
      modal.classList.remove("open");
      modal.setAttribute("aria-hidden", "true");
      document.body.classList.remove("modal-open");
    }

    function closeAllModals() {
      document.querySelectorAll(".modal-wrap.open").forEach(modal => {
        modal.classList.remove("open");
        modal.setAttribute("aria-hidden", "true");
      });
      document.body.classList.remove("modal-open");
    }

    function backdropClose(event, id) {
